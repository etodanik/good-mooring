cmake_minimum_required(VERSION 3.25)
project(MooringTracyTools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(TRACY_SOURCE_DIR "" CACHE PATH "Pinned Tracy 0.14.1 source directory")
if(NOT EXISTS "${TRACY_SOURCE_DIR}/capture/CMakeLists.txt")
    message(FATAL_ERROR "Set TRACY_SOURCE_DIR to the Tracy source fetched by the app build")
endif()
add_subdirectory("${TRACY_SOURCE_DIR}/capture" capture)
add_executable(mooring-tracy-inspect Inspect.cpp)
target_link_libraries(mooring-tracy-inspect PRIVATE TracyServer)
